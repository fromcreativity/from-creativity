<script id="fc-life-motion">
(function(){
 'use strict';
 var watcher, expiry;
 function start(){
  var root=document.getElementById('fc-life-documentary');
  if(!root)return false;
  if(watcher)watcher.disconnect();
  clearTimeout(expiry);
  if(root.dataset.lifeMotion)return true;
  root.dataset.lifeMotion='20261004';
  var reduced=window.matchMedia('(prefers-reduced-motion: reduce)'), observer;
  if(reduced.matches||!Element.prototype.animate)return true;
  function move(el,delay,distance){
   if(!el||reduced.matches||!el.isConnected)return;
   el.animate([{opacity:.62,transform:'translateY('+distance+'px)'},{opacity:1,transform:'translateY(0)'}],{duration:850,delay:delay||0,easing:'cubic-bezier(.2,.7,.3,1)',fill:'backwards'});
  }
  if(!location.hash){
   root.querySelectorAll('h1 span').forEach(function(el,i){move(el,i*120,18)});
   move(root.querySelector('.life-hero-copy'),200,14);
  }
  if('IntersectionObserver' in window){
   observer=new IntersectionObserver(function(entries){entries.forEach(function(entry){if(entry.isIntersecting){move(entry.target,0,16);observer.unobserve(entry.target)}})},{threshold:.12,rootMargin:'0px 0px -32px 0px'});
   root.querySelectorAll('.life-section-label,.life-section-body h2,.life-portrait,.life-context,.life-contact h2').forEach(function(el){observer.observe(el)});
  }
  function stop(){if(!reduced.matches)return;if(observer)observer.disconnect();root.getAnimations({subtree:true}).forEach(function(animation){animation.cancel()})}
  if(reduced.addEventListener)reduced.addEventListener('change',stop);
  return true;
 }
 function boot(){if(!start()){
   watcher=new MutationObserver(start);
   watcher.observe(document.documentElement,{childList:true,subtree:true});
   expiry=setTimeout(function(){watcher.disconnect()},10000);
  }
 }
 if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot,{once:true});else boot();
})();
</script>
